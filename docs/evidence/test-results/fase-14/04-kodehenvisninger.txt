# Treffene er sjekket for hånd: navn som er med vilje fraværende (product_id, source_recipe_id, actorName, singletonKey), pakkenavn og regler (@fastify/static, zod/mini, react-hooks/refs), og ting utenfor repoet (ghostweb, noindex i Caddy). bytea finnes som BYTEA i migreringen.

import re,os,subprocess,sys
files=sys.argv[1:]
tracked=subprocess.run(['git','ls-files'],capture_output=True,text=True).stdout.split('\n')
alltext=None
def grep(tok):
    r=subprocess.run(['git','grep','-lF','--',tok,'--','apps','packages','prisma','e2e','docker','scripts','package.json'],capture_output=True,text=True)
    return r.returncode==0
for f in files:
    s=open(f,encoding='utf-8').read()
    toks=sorted(set(re.findall(r'`([^`\n]+)`',s)))
    miss_path=[];miss_id=[]
    for t in toks:
        if '/' in t and re.match(r'^[\w@.\-/\[\]]+$',t) and not t.startswith('/'):
            base=t.rstrip('/')
            if not (os.path.exists(base) or any(x.endswith(base) or ('/'+base+'/') in ('/'+x) for x in tracked if x)):
                miss_path.append(t)
        elif re.match(r'^[A-Za-z_][\w.]{3,}(\(\))?$',t):
            name=t.replace('()','')
            if not grep(name.split('.')[-1]): miss_id.append(t)
    print(f'== {f}: {len(toks)} kodebiter, stier som ikke finnes: {miss_path}\n   navn som ikke finnes i koden: {miss_id}')
