# Skriptet:
import re,glob
def key(s): return (int(re.search(r'fase-(\d+)',s).group(1)),s)
for f in sorted(glob.glob('docs/evidence/test-results/fase-*/10-mutation-check*.txt'),key=key):
    lines=re.sub(r'\x1b\[[0-9;]*m','',open(f,encoding='utf-8',errors='replace').read()).splitlines()
    rounds=[];cur=None
    for l in lines:
        m=re.match(r'^ ?(\d+)\. (.*)',l)
        if m: cur=[m.group(1),m.group(2),False,False]; rounds.append(cur); continue
        if l.startswith('Uendret kode'): cur=None; continue
        if cur:
            if re.search(r'\d+ failed|feilet:',l): cur[2]=True
            if 'FORVENTET' in l: cur[3]=True
    rounds=[r for r in rounds if not r[1].startswith('Uendret kode')]
    c=sum(r[2] for r in rounds)
    print(f.split('test-results/')[1], f'{c}/{len(rounds)}', [r[0]+('*' if r[3] else '') for r in rounds if not r[2]])
